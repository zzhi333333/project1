X = np.array([[1.2, 0.5], [0.8, 0.3], [1.5, 0.7], [1.0, 0.4], [0.7, 0.2], [1.3, 0.6], [1.1, 0.45], [0.9, 0.35], [1.4, 0.65], [1.6, 0.75]])
y = np.array([1, 1, 2, 1, 1, 2, 1, 1, 2, 3])  # 将y从one-hot编码改为整数形式，方便定义奖励函数

# 马尔科夫决策过程模型
class MarkovDecisionProcess():
    def __init__(self, n_states, n_actions, transition_probs, rewards, gamma=0.9, epsilon=0.1):
        self.n_states = n_states  # 状态数
        self.n_actions = n_actions  # 动作数
        self.transition_probs = transition_probs  # 状态转移概率矩阵
        self.rewards = rewards  # 奖励矩阵
        self.gamma = gamma  # 折扣因子
        self.epsilon = epsilon  # 探索因子
        self.values = np.zeros(n_states)  # 状态值函数

    def update(self, state):
        q_values = []
        for action in range(self.n_actions):
            q_value = 0
            for next_state in range(self.n_states):
                q_value += self.transition_probs[state, action, next_state] * (self.rewards[state, action, next_state] + self.gamma * self.values[next_state])
            q_values.append(q_value)
        best_action = np.argmax(q_values)
        return best_action

    def train(self, n_epochs=1000):
        for epoch in range(n_epochs):
            state = np.random.randint(self.n_states)  # 随机选择初始状态
            done = False
            while not done:
                action = self.update(state)
                prob = np.ones(self.n_actions) * self.epsilon / self.n_actions  # 加入探索因子
                prob[action] += 1 - self.epsilon
                next_state = np.random.choice(self.n_states, p=self.transition_probs[state, action])
                reward = self.rewards[state, action, next_state]
                self.values[state] = reward + self.gamma * np.max(self.values)  # 更新状态值函数
                state = next_state
                if epoch == n_epochs - 1:  # 输出最后一个epoch中的状态值函数
                    print(f"State {state}: value {self.values[state]}")
                if reward > 0:  # 预测正确，获得正向奖励
                    done = True
                elif reward < 0:  # 预测错误，获得负向奖励
                    done = True
                else:  # 继续尝试
                    done = False
