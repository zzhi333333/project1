import pandas as pd
import numpy as np
import tkinter as tk
from tkinter import filedialog
from tkinter import messagebox


# module1 MPS 文档导入生成透视表 #
root = tk.Tk()
root.withdraw()

messagebox.showinfo("Notice", "Please select MPS1.xls!")
Filepath = filedialog.askopenfilename()

messagebox.showinfo("Notice", "Please select MPS2.xls file!")
Filepath1 = filedialog.askopenfilename()

LNB_MPS1 = Filepath
LNB_MPS2 = Filepath1

df1=pd.read_excel(LNB_MPS1, sheet_name=0, header=1)
df2=pd.read_excel(LNB_MPS2, sheet_name=0, header=1)

MPS1_Pivot=pd.pivot_table(df1, index = ['odm','subgeo','series'],
                       values = ['May','Jun','Jul','Aug','Sep','Oct','Nov'], aggfunc = np.sum)
MPS1_Pivot_flat=MPS1_Pivot.reset_index()
MPS1_Pivot_flat=MPS1_Pivot_flat.reindex(['odm','subgeo','series','May','Jun','Jul','Aug','Sep','Oct','Nov'], axis=1)

MPS2_Pivot=pd.pivot_table(df2, index = ['odm','subgeo','series'],
                       values = ['May','Jun','Jul','Aug','Sep','Oct','Nov'], aggfunc = np.sum)
MPS2_Pivot_flat=MPS2_Pivot.reset_index()
MPS2_Pivot_flat=MPS2_Pivot_flat.reindex(['odm','subgeo','series','May','Jun','Jul','Aug','Sep','Oct','Nov'], axis=1)


MPS1_Pivot_flat.to_excel('D:\Python\W2W FCST\MPS1_Piovt.xlsx', index=False)
MPS2_Pivot_flat.to_excel('D:\Python\W2W FCST\MPS2_Piovt.xlsx', index=False)

#module2， W2跟W1比对差值，需要注意填充空值部分
W2W_result = pd.merge(MPS2_Pivot_flat, MPS1_Pivot_flat[['odm','subgeo', 'series', 'May','Jun','Jul','Aug','Sep','Oct','Nov']],
                  how='outer', on = ['odm','subgeo', 'series'], sort = False)

W2W_result=W2W_result.fillna(0) #空值NAN填充为0

W2W_result['May_w2w']=W2W_result['May_x']-W2W_result['May_y']
W2W_result['Jun_w2w']=W2W_result['Jun_x']-W2W_result['Jun_y']
W2W_result['Jul_w2w']=W2W_result['Jul_x']-W2W_result['Jul_y']
W2W_result['Aug_w2w']=W2W_result['Aug_x']-W2W_result['Aug_y']
W2W_result['Sep_w2w']=W2W_result['Sep_x']-W2W_result['Sep_y']
W2W_result['Oct_w2w']=W2W_result['Oct_x']-W2W_result['Oct_y']
W2W_result['Nov_w2w']=W2W_result['Nov_x']-W2W_result['Nov_y']

W2W_result.to_excel('D:\Python\W2W FCST\W2W_result.xlsx',index=False)
