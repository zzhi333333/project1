# module1 PSD匹配POR #
root = tk.Tk()

messagebox.showinfo("Notice", "Please select LNB PSD.xlsx file!")
Filepath = filedialog.askopenfilename()

messagebox.showinfo("Notice", "Please select POR1.xlsx file!")
Filepath1 = filedialog.askopenfilename()


LNB_PSD = Filepath
LNB_POR = Filepath1

df1 = pd.DataFrame(pd.read_excel(LNB_PSD, sheet_name = 0)) # 读取PSD #
df2 = pd.DataFrame(pd.read_excel(LNB_POR, sheet_name = 0)) # 读取POR1 #
df3 = df2.drop_duplicates(['PO_ITEM']) # POR1 针对PO+ITEM 自动去重 #

# 按MTM字段匹配POR信息，PSD为left， POR为right， 用left 方式，df2中的字段可增减，如果不指定列名字典，即为POR中全部的信息 #
# 注意如果要切片选择字段，连接词必须包括在内#
result = pd.merge(df1, df3[['PO_ITEM', 'Battery', 'Camera', 'Hard_Drive', 'SSD', 'Display_Type',\
                                  'VGA', 'EMMC', 'Processor_AMD', 'Processor_CP']], \
                  how='left', on = 'PO_ITEM', sort = False)

writer = pd.ExcelWriter('D:/Python/Compal ATP project/PSD POR file/CPL_PSD_with_POR.xlsx')
result.to_excel(writer,index=False)
writer.save()

print('CPL_PSD_with_POR is successfully exported!')

# module2, 从PSD_with_POR 文件中抓取FRSD订单行，写入FRSD_PSD_with_POR 文件 #

# 获取月底日期，用YYYY-MM-DD格式输出 #
now = datetime.datetime.now()
year = now.year
month = now.month
day = calendar.monthrange(year, month)
last_day = calendar.monthrange(year, month)[1] # 输出一个月的第一天和最后一天数字的元组， [1]取元组第二个值， 即当月最后一天 #
month_end = datetime.date(year, month, last_day)

LNB_PSD1 = 'D:/Python/Compal ATP project/PSD POR file/CPL_PSD_with_POR.xlsx'
df4 = pd.DataFrame(pd.read_excel(LNB_PSD1, sheet_name = 0))

FRSD_PSD = df4.loc[df4['RSD'] > str(month_end)]  # 提取FRSD订单行，调整为str，获取当月月底时间代码如上，不需要手动输入 #

writer = pd.ExcelWriter('D:/Python/Compal ATP project/PSD POR file/CPL_FRSD_PSD_with_POR.xlsx')
FRSD_PSD.to_excel(writer, index = False)
writer.save()

print('FRSD lines are successfully exported!')
