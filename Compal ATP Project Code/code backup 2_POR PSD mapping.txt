import pandas as pd
import numpy as np
import calendar
import datetime
import tkinter as tk
from tkinter import filedialog
from tkinter import messagebox

# module1 PSD匹配POR #
root = tk.Tk()
root.withdraw()


messagebox.showinfo("Notice", "Please select LNB_PSD.xlsx file!")
Filepath = filedialog.askopenfilename()

messagebox.showinfo("Notice", "Please select POR1.xlsx file!")
Filepath1 = filedialog.askopenfilename()


LNB_PSD = Filepath
LNB_POR = Filepath1

df1 = pd.DataFrame(pd.read_excel(LNB_PSD,sheet_name='PSD')) # 读取PSD #

df2 = pd.DataFrame(pd.read_excel(LNB_POR,sheet_name=0)) # 读取POR1 #


# 按MTM字段匹配POR信息，PSD为left， POR为right， 用left 方式，df2中的字段可增减，如果不指定列名字典，即为POR中全部的信息 #
# 注意如果要切片选择字段，连接词必须包括在内#
result = pd.merge(df1, df2[['MTM', 'Battery', 'Camera', 'Hard_Drive', 'SSD', 'Display_Type', \
                                  'EMMC', 'Processor_AMD', 'Processor_CP']], \
                  how='left', on = 'MTM', sort = False)

result.to_excel('D:\Python projects\Compal ATP Project Code\CPL_PSD_with_POR.xlsx')

print('CPL_PSD_with_POR is successfully exported!')
